        "def find_column(df, keywords, preferred_exact=None):\n",
        "    cols = list(df.columns)\n",
        "    lower = {str(c).lower(): c for c in cols}\n",
        "\n",
        "    if preferred_exact:\n",
        "        for p in preferred_exact:\n",
        "            if p.lower() in lower:\n",
        "                return lower[p.lower()]\n",
        "\n",
        "    for c in cols:\n",
        "        cl = str(c).lower()\n",
        "        if all(k.lower() in cl for k in keywords):\n",
        "            return c\n",
        "\n",
        "    return None\n",
        "\n",
        "country_col = find_column(index_df, [\"country\"]) or find_column(index_df, [\"name\"])\n",
